import random

WIDTH = 600
HEIGHT = 450

TITLE = "Aventura espacial"
FPS = 30

# Objetos e variáveis
ship = Actor("ship", (300, 400), size = (50,50))
space = Actor("space")
space2 = Actor("space2", size = (600,450))
enemies = []
meteors = []
bullets = []
inimigos= ["enemy", "inimigo2", "inimigo3"]
mode = 'menu'
type1 = Actor("ship", (150, 270), size = (50,50))
type2 = Actor("ship-2", (300, 270), size = (50,50))
type3 = Actor("ship-3", (450, 270), size = (50,50))
count = 0
contagem = 0
# Preenchendo a lista de inimigos
for i in range(5):
    x = random.randint(0, 600)
    y = random.randint(-450, -50)
    imagem = random.choice(inimigos)
    enemy = Actor(imagem, (x, y), size = (45,45))
    enemy.speed = random.randint(4, 10)
    enemies.append(enemy)
    
# Preenchendo a lista de meteoros
for i in range(5):
    x = random.randint(0, 600)
    y = random.randint(-450, -50)
    meteor = Actor("meteor", (x, y), size = (30,30))
    meteor.speed = random.randint(2, 10)
    meteors.append(meteor)

# Desenhando os gráficos
def draw():
   # Estado do jogo
    if mode == 'menu':
        space2.draw()
        screen.draw.text('Escolha sua nave', center = (300, 185), color = "white", fontsize = 32)
        type1.draw()
        type2.draw()
        type3.draw()
    if mode == 'game':
        space.draw()
        
        
        # Resultado da pontuação
        screen.draw.text(count, center = (20, 10), color = "white", fontsize = 24)
        
        # Desenhando os meteoros
        for i in range(len(meteors)):
            meteors[i].draw()
        ship.draw()
        # Desenhando os inimigos
        for i in range(len(enemies)):
            enemies[i].draw()
        # Desenhando as balas
        for i in range(len(bullets)):
            bullets[i].draw()
        
    # Janela de Game Over   
    elif mode == 'end':
        space.draw()
        screen.draw.text("GAME OVER!", center = (300, 200), color = "white", fontsize = 36)
        #Pontuação final
screen.draw.text(count, center = (10, 10), color = "white", fontsize = 24)
    
# Controles
def on_mouse_move(pos):
    ship.pos = pos

# Adicionando um novo inimigo à lista
def new_enemy():
    x = random.randint(0, 400)
    y = -50
    imagem = random.choice(inimigos)
    enemy = Actor(imagem, (x, y), size = (45,45))
    enemy.speed = random.randint(4, 10)
    enemies.append(enemy)

# Movimento do inimigo 
def enemy_ship():
    global contagem
    for i in range(len(enemies)):
        if enemies[i].y < 650:
            enemies[i].y = enemies[i].y + enemies[i].speed
        else:
            enemies.pop(i)
            contagem += 1
            new_enemy()

# Movimento dos planetas


# Movimento dos meteoros
def meteorites():
    for i in range(len(meteors)):
        if meteors[i].y < 450:
            meteors[i].y = meteors[i].y + meteors[i].speed
        else:
            meteors[i].x = random.randint(0, 600)
            meteors[i].y = -20
            meteors[i].speed = random.randint(2, 10)

# Colisões
def collisions():
    global mode
    global count
    for i in range(len(enemies)):
        if ship.colliderect(enemies[i]):
            mode = 'end'
            count = 0
        # Colisões com balas
        for j in range(len(bullets)):
           if bullets[j].colliderect(enemies[i]):
                enemies.pop(i)
                bullets.pop(j)
                new_enemy()
                count += 1
                break 

def update(dt):
    global mode, contagem
    if mode == 'game':
        enemy_ship()
        collisions()
        meteorites()
        if contagem >= 10:
            mode = "end"
        # Movimento das balas
        for i in range(len(bullets)):
            if bullets[i].y < 0:
                bullets.pop(i)
                break
            else:
                bullets[i].y = bullets[i].y - 10

    elif mode == "end" and keyboard.space:
        mode = "game"
        contagem = 0
    
        

# Função de manuseio de cliques
def on_mouse_down(button, pos):
    global mode
    global ship
    if mode == 'menu' and type1.collidepoint(pos):
        ship.image = "ship"
        mode = 'game'
    elif mode == 'menu' and type2.collidepoint(pos):
        ship.image = "ship-2"
        mode = 'game'
    elif mode == 'menu' and type3.collidepoint(pos):
        ship.image = "ship-3"
        mode = 'game'
    # Mecânicas de tiro
    elif mode == 'game' and button == mouse.LEFT:
        bullet = Actor("bullet", size = (20,20))
        bullet.pos = ship.pos
        bullets.append(bullet)
